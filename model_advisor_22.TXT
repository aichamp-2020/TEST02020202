#!/usr/bin/env python3
"""
Model Advisor — pick the right LLM for an insurance (or general) use case.

Companion tool to "Cross-Vendor Model Comparison, Insurance Use Cases &
Recommendations". Same model database and scoring logic as that deck,
wrapped in a simple chat interface so you can type a use case in plain
English and get a ranked recommendation with reasoning.

Usage:
    python model_advisor.py                  # interactive chat mode
    python model_advisor.py "FNOL triage"     # one-shot mode, single query
    python model_advisor.py --list            # show all known use cases
    python model_advisor.py --models          # show the full model table

No external dependencies — standard library only.
"""

from __future__ import annotations
import sys
import textwrap
import argparse
from dataclasses import dataclass, field
from typing import Optional


# ============================================================
# MODEL DATABASE
# Same 13 models and figures used in the cross-vendor comparison deck.
# Pricing per 1M tokens (USD). Context in tokens. Intelligence Index
# from Artificial Analysis where independently measured; otherwise None.
# Source: vendor API pricing pages + Artificial Analysis, accessed Jun 2026.
# ============================================================

@dataclass
class Model:
    name: str
    vendor: str
    input_price: float       # $ per 1M input tokens
    output_price: float      # $ per 1M output tokens
    cache_hit_price: Optional[float]  # $ per 1M cached input tokens, if published
    context: int              # max context window, tokens
    intelligence: Optional[int]    # Artificial Analysis Intelligence Index, or None
    speed: int                 # approx output tokens/sec
    ttft: float                # approx time-to-first-token, seconds
    knowledge_cutoff: str       # human-readable reliable knowledge cutoff
    release_date: str           # human-readable release date
    eval_cost: Optional[float]  # $ to run full Artificial Analysis Intelligence Index suite, if published
    reasoning: bool            # is this a reasoning-tuned model
    vision: bool               # native multimodal / vision input support
    open_weight: bool          # open weights available
    ocr_accuracy: Optional[float] = None   # measured doc/OCR accuracy %, if known
    audio_input: bool = False    # native audio understanding (not just transcription bolted on)
    video_input: bool = False    # native video understanding (frame/clip analysis)
    image_gen: bool = False      # vendor offers native/sibling image generation in the same ecosystem
    notes: str = ""             # one-line vendor-level positioning note

    def blended_price(self, cache_ratio=0.7, in_ratio=0.2, out_ratio=0.1) -> float:
        """
        Blended $/M tokens using Artificial Analysis's standard 7:2:1
        cache-hit:input:output weighting, which approximates a typical
        production traffic mix (most tokens are repeated context).
        Falls back to input price for the cache-hit share if no
        published cache price exists.
        """
        cache = self.cache_hit_price if self.cache_hit_price is not None else self.input_price
        return cache * cache_ratio + self.input_price * in_ratio + self.output_price * out_ratio

    def justify(self, param: str) -> str:
        """Return a short justification for one parameter value, used by
        the --explain flag and the reference report."""
        j = JUSTIFICATIONS.get((self.name, param))
        if j:
            return j
        return PARAM_FALLBACK_JUSTIFICATIONS.get(param, "No specific justification on file for this parameter.")

# ============================================================
# SCORING LOGIC
# Identical to the logic implemented in the in-chat recommendation
# artifact and described in the deck (Section 6.1 / addendum Section 3.1).
# ============================================================

INTEL_TARGET_BY_COMPLEXITY = {1: 15, 2: 25, 3: 35, 4: 45, 5: 55}
CONTEXT_FLOOR_BY_LEVEL = {1: 8_000, 2: 32_000, 3: 200_000, 4: 1_000_000, 5: 2_000_000}


def score_model(m: Model, uc: UseCase) -> float:
    s = 0.0

    # 1. Complexity -> Intelligence target. Under-provisioning is penalized
    #    harder than over-provisioning (quality risk costs more than headroom).
    intel = m.intelligence if m.intelligence is not None else INTEL_TARGET_BY_COMPLEXITY[3]
    target = INTEL_TARGET_BY_COMPLEXITY[uc.complexity]
    s -= abs(intel - target) * 1.3
    if intel < target - 10:
        s -= 25

    # 2. Latency -> speed and time-to-first-token. A baseline TTFT penalty applies
    #    at every latency level (an outlier-slow TTFT, e.g. >10s, is a real cost
    #    even for batch-tolerant workloads — timeouts, poor debugging UX, queuing
    #    backpressure) and scales up sharply as latency sensitivity increases.
    s -= m.ttft * 0.8
    if uc.latency >= 4:
        s += min(m.speed / 4, 40) - m.ttft * 3

    # 3. Volume -> cost weighting. High volume penalizes expensive models harder.
    if uc.volume >= 4:
        s -= (m.output_price / 2 + m.input_price) * 2.5

    # 4. Context -> hard filter, then a small bonus for headroom.
    ctx_need = CONTEXT_FLOOR_BY_LEVEL[uc.context]
    if m.context < ctx_need:
        s -= 45
    else:
        s += min((m.context - ctx_need) / 500_000, 6)

    # 5. Regulated workflow -> soft penalty for non-reasoning models at
    #    complexity 3+, small bonus for Anthropic (benchmarked low
    #    hallucination rates make it a reasonable regulated default).
    if uc.regulated and not m.reasoning and uc.complexity >= 3:
        s -= 10
    if uc.regulated and m.vendor == "Anthropic":
        s += 6

    # 6. Document modality -> vision is a hard requirement if needed.
    #    Measured OCR/doc accuracy is weighted heavily here: a misread
    #    field on a real document costs more than the token savings, so
    #    this is intentionally allowed to outweigh the volume/cost
    #    penalty above. Models with no published accuracy figure get a
    #    small caution penalty rather than a free pass relative to
    #    models that *do* publish a number.
    if uc.vision:
        if not m.vision:
            s -= 100
        elif m.ocr_accuracy:
            s += (m.ocr_accuracy - 85) * 2.2
        else:
            s -= 6

    # 7. Open weights -> hard requirement if self-hosting is needed.
    if uc.open_weight and not m.open_weight:
        s -= 60

    # 8. Audio -> hard requirement for native audio understanding. This is
    #    a real capability gap, not a quality spectrum: as of this data,
    #    only Gemini 3.1 (Pro/Flash-Lite) and Llama 4 Maverick offer it.
    #    A model without it cannot do the job at all, regardless of how
    #    well it scores on every other axis.
    if uc.audio and not m.audio_input:
        s -= 120

    # 9. Video -> hard requirement for native video understanding (frame/
    #    motion analysis in one call, not a pre-extracted-frames workaround).
    if uc.video and not m.video_input:
        s -= 120

    # 10. Image generation -> hard requirement if the use case needs the
    #     vendor to produce images, not just read them.
    if uc.image_gen and not m.image_gen:
        s -= 80

    return s


def rank_models(uc: UseCase) -> list[tuple[Model, float]]:
    ranked = [(m, score_model(m, uc)) for m in MODELS]
    ranked.sort(key=lambda pair: pair[1], reverse=True)
    return ranked


def find_model_by_name(query: str) -> Optional[Model]:
    """
    Look up a single model by name, case-insensitive. Tries, in order:
    exact match, then "starts with", then substring — so a query like
    "GPT-5.4" resolves to GPT-5.4 itself rather than the first model whose
    name happens to contain that text (e.g. "GPT-5.4 mini").
    """
    q = query.lower().strip()
    for m in MODELS:
        if m.name.lower() == q:
            return m
    for m in MODELS:
        if m.name.lower().startswith(q):
            return m
    for m in MODELS:
        if q in m.name.lower():
            return m
    return None

# USE CASE MATCHING (free-text -> UseCase)
# ============================================================

def find_use_case(query: str) -> Optional[UseCase]:
    """Find the best-matching known use case for a free-text query."""
    q = query.lower().strip()
    if not q:
        return None

    # Exact name match first.
    for uc in USE_CASES:
        if uc.name.lower() == q:
            return uc

    # Keyword / substring match, scored by overlap.
    best, best_score = None, 0
    for uc in USE_CASES:
        haystack = [uc.name.lower()] + uc.keywords
        score = 0
        for term in haystack:
            if term in q or q in term:
                score += len(term)
        if score > best_score:
            best, best_score = uc, score

    return best


def guess_use_case_from_description(query: str) -> UseCase:
    """
    Fallback: if no known use case matches, build a rough UseCase from
    simple heuristics on the free-text description so the tool still
    returns something useful instead of failing.
    """
    q = query.lower()

    def has_any(*words):
        return any(w in q for w in words)

    complexity = 3
    if has_any("simple", "basic", "lookup", "classify", "classification", "routing"):
        complexity = 2
    if has_any("reasoning", "complex", "analysis", "investigat", "fraud", "legal", "compliance", "regulat"):
        complexity = 4
    if has_any("highest stakes", "bias", "discrimination", "litigation", "audit"):
        complexity = 5

    latency = 3
    if has_any("real time", "real-time", "chat", "instant", "live"):
        latency = 5
    if has_any("batch", "overnight", "offline", "report"):
        latency = 1

    volume = 3
    if has_any("high volume", "scale", "millions", "thousands of", "every customer"):
        volume = 4
    if has_any("low volume", "occasional", "rare", "few"):
        volume = 1

    context = 2
    if has_any("long document", "large document", "treaty", "full file", "entire", "huge", "big context"):
        context = 4
    if has_any("short", "single field", "one line"):
        context = 1

    regulated = has_any("regulat", "compliance", "audit", "legal", "underwrit", "pricing", "bias",
                         "discrimination", "medical", "claim denial", "rate")
    vision = has_any("image", "photo", "scan", "pdf", "ocr", "document", "invoice", "form", "imagery")
    audio = has_any("audio", "voice", "call recording", "transcri", "speech", "phone call", "ivr")
    video = has_any("video", "footage", "drone", "walkthrough", "recording of")
    image_gen = has_any("generate an image", "generate image", "create an image", "image generation",
                         "marketing visual", "graphic design", "ad creative", "mockup image")
    open_weight = has_any("self-host", "self host", "on-prem", "on prem", "data residency", "open weight", "open-weight")

    return UseCase(
        name=f'Custom: "{query.strip()}"',
        category="Ad hoc",

    )


# ============================================================
# PTU (PROVISIONED THROUGHPUT) VS. PAY-AS-YOU-GO CALCULATOR
#
# Azure OpenAI (and a growing set of Foundry-hosted models) offers a second
# pricing mode alongside per-token billing: Provisioned Throughput Units
# (PTUs). A PTU is reserved capacity billed hourly — $/PTU/hour — regardless
# of how many tokens you actually process. Whether that's cheaper than
# per-token (PAYG) pricing depends entirely on your volume and how much of
# that reserved capacity you actually use (utilization).
#
# Three things genuinely vary by model and deployment and are NOT guessed
# here — they're left as configurable inputs with a clearly-labeled default:
#   - tpm_per_ptu: tokens-per-minute one PTU can deliver for this model.
#     Microsoft does not publish one universal number; it depends on the
#     model and your call shape (prompt/response size mix). The figure
#     supplied here is only ever a planning default — verify with Azure's
#     own capacity calculator before committing budget.
#   - ptu_hourly_cost: $/PTU/hour, which varies by region, model, and
#     whether you're on hourly billing vs. a monthly/annual reservation.
#   - output_weight: how many "input-token-equivalents" one output token
#     consumes against PTU capacity (NOT the same as the $ price ratio).
#     Microsoft states this generally tracks the model's global standard
#     output:input price ratio for GPT-4.1+ models, so that ratio is used
#     as the default, but it is an approximation, not a published constant.
#
# Source: Microsoft Learn, "Provisioned throughput for Foundry Models" and
# "Provisioned throughput unit (PTU) costs and billing" (accessed Jun 2026).
# ============================================================

# Reported planning-default hourly PTU costs, by billing commitment. These
# are industry-reported approximate figures (Microsoft does not publish a
# flat list price), not a quote — always confirm current pricing for your
# region and agreement type before budgeting.
PTU_COST_DEFAULTS = {
    "hourly": 3.35,     # ~$2,448/month / (730 hrs) on-demand hourly billing
    "monthly_reservation": 2.02,   # ~30-40% discount vs hourly, reported
    "annual_reservation": 1.31,    # ~60-70% discount vs hourly, reported
}

# Planning-default TPM-per-PTU figures. These vary by model, region, and
# call shape — treat as a rough starting point only, per Microsoft's own
# guidance to validate with their capacity calculator before committing.
TPM_PER_PTU_DEFAULTS = {
    "Claude Haiku 4.5": 2_500, "Claude Sonnet 4.6": 2_000, "Claude Opus 4.7": 1_200,
    "GPT-5.4 mini": 3_000, "GPT-4.1": 2_800, "GPT-5.4": 2_500, "GPT-5.5": 2_200,
    "Gemini 3.1 Flash-Lite": 3_500, "Gemini 3.1 Pro": 2_200,
    "Llama 4 Scout": 2_800, "Llama 4 Maverick": 2_400,
    "DeepSeek V3.2": 2_600, "DeepSeek R1": 2_000,
    "Mistral Large 3": 2_300, "Phi-4": 4_000,
    "Command A": 2_400, "Command A+": 2_600,
    "Qwen3.7 Max": 2_200, "Grok 4.3": 2_000,
    "Nova Pro": 2_400, "Nova Micro": 4_000,
    "Gemma 4 E4B": 3_800, "Llama 3.2 3B": 4_200, "Qwen3.5 9B": 2_900,
    "Ministral 3 8B": 3_400, "Granite 4.1 8B": 3_600,
}
DEFAULT_TPM_PER_PTU_FALLBACK = 2_500  # used for any model not in the table above


@dataclass
class PTUConfig:
    """User-configurable PTU parameters. Every field can be overridden;
    defaults are clearly-labeled planning estimates, not vendor quotes."""
    ptus_available: float            # number of PTUs you have or are pricing out
    ptu_hourly_cost: float            # $ per PTU per hour
    tpm_per_ptu: Optional[float] = None   # tokens/minute one PTU delivers; None -> use default
    hours_per_month: float = 730.0    # billing assumes 24/7 reservation by default
    output_weight: Optional[float] = None  # output-token capacity multiplier; None -> derive from price ratio


def ptu_capacity_tpm(model: Model, cfg: PTUConfig) -> float:
    """Total tokens-per-minute capacity this PTU configuration delivers."""
    tpm = cfg.tpm_per_ptu if cfg.tpm_per_ptu is not None else TPM_PER_PTU_DEFAULTS.get(model.name, DEFAULT_TPM_PER_PTU_FALLBACK)
    return tpm * cfg.ptus_available


def ptu_monthly_cost(cfg: PTUConfig) -> float:
    """Total $ cost of the reserved PTU capacity for one month, regardless of usage."""
    return cfg.ptus_available * cfg.ptu_hourly_cost * cfg.hours_per_month


def payg_monthly_cost(model: Model, monthly_input_tokens: float, monthly_output_tokens: float) -> float:
    """Standard per-token (pay-as-you-go) cost for the same monthly volume."""
    return (monthly_input_tokens / 1_000_000) * model.input_price + \
           (monthly_output_tokens / 1_000_000) * model.output_price


def compare_ptu_vs_payg(model: Model, cfg: PTUConfig,
                         monthly_input_tokens: float, monthly_output_tokens: float) -> dict:
    """
    Full PTU-vs-PAYG comparison for a given model, PTU configuration, and
    expected monthly volume. Returns both directions: whether the supplied
    PTU capacity covers the volume, and which option is cheaper either way.
    """
    out_weight = cfg.output_weight if cfg.output_weight is not None else (model.output_price / model.input_price if model.input_price else 4.0)

    # Required capacity, expressed in input-token-equivalents per minute.
    total_tokens = monthly_input_tokens + monthly_output_tokens
    weighted_tokens_per_month = monthly_input_tokens + monthly_output_tokens * out_weight
    minutes_per_month = cfg.hours_per_month * 60
    required_tpm = weighted_tokens_per_month / minutes_per_month if minutes_per_month else 0

    available_tpm = ptu_capacity_tpm(model, cfg)
    covers_volume = available_tpm >= required_tpm
    utilization = (required_tpm / available_tpm) if available_tpm else float("inf")

    ptu_cost = ptu_monthly_cost(cfg)
    payg_cost = payg_monthly_cost(model, monthly_input_tokens, monthly_output_tokens)

    ptus_needed = (required_tpm / (cfg.tpm_per_ptu if cfg.tpm_per_ptu is not None
                   else TPM_PER_PTU_DEFAULTS.get(model.name, DEFAULT_TPM_PER_PTU_FALLBACK))) if required_tpm else 0

    cheaper = "PTU" if ptu_cost < payg_cost else "PAYG"
    savings = abs(ptu_cost - payg_cost)
    savings_pct = (savings / max(ptu_cost, payg_cost) * 100) if max(ptu_cost, payg_cost) else 0

    # A 0-PTU configuration provides zero capacity, not a "free win" — don't
    # let the cost comparison claim PTU is cheaper when it can't serve any
    # traffic at all. Treat this as PAYG being the only viable option.
    if cfg.ptus_available <= 0:
        cheaper = "PAYG"
        savings = payg_cost
        savings_pct = 100.0 if payg_cost > 0 else 0.0

    return {
        "model": model.name,
        "required_tpm": required_tpm,
        "available_tpm": available_tpm,
        "covers_volume": covers_volume,
        "utilization_pct": utilization * 100 if utilization != float("inf") else None,
        "ptus_needed": ptus_needed,
        "ptu_monthly_cost": ptu_cost,
        "payg_monthly_cost": payg_cost,
        "cheaper_option": cheaper,
        "monthly_savings": savings,
        "savings_pct": savings_pct,
        "output_weight_used": out_weight,
        "tpm_per_ptu_used": cfg.tpm_per_ptu if cfg.tpm_per_ptu is not None else TPM_PER_PTU_DEFAULTS.get(model.name, DEFAULT_TPM_PER_PTU_FALLBACK),
    }


def print_ptu_comparison(model: Model, cfg: PTUConfig, monthly_input_tokens: float, monthly_output_tokens: float) -> None:

          f"{r['tpm_per_ptu_used']:,.0f} TPM/PTU "
          f"{'(your value)' if cfg.tpm_per_ptu is not None else '(planning default — verify with Azure capacity calculator)'}")
    print(f"  Output-token capacity weight: {r['output_weight_used']:.2f}x "
          f"{'(your value)' if cfg.output_weight is not None else '(derived from output:input price ratio — approximation)'}")
    print()
    print(f"  Required throughput:  {r['required_tpm']:,.0f} TPM")
    print(f"  Your PTU capacity:    {r['available_tpm']:,.0f} TPM")
    if cfg.ptus_available <= 0:
        print("  -> 0 PTUs provides no capacity at all — this isn't a real deployment option, just a reference point.")
    elif r["covers_volume"]:
        print(f"  -> Your PTUs COVER this volume, at {r['utilization_pct']:.1f}% utilization.")
    else:
        util_str = f"{r['utilization_pct']:.1f}%" if r["utilization_pct"] is not None else "undefined (zero PTU capacity)"

              f"({util_str} of capacity needed). "

    print(f"  PTU monthly cost:   {fmt_money(r['ptu_monthly_cost'])}  (fixed, regardless of usage)")

        print(f"  -> With 0 PTUs there's no PTU option to compare — PAYG is the only choice "
              f"at {fmt_money(r['payg_monthly_cost'])}/month for this volume.")

        print("  Caution: utilization is under 50% — PTU often only wins at sustained")
        print("  high utilization. Re-check this once you have real traffic data.")
    if not r["covers_volume"] and cfg.ptus_available > 0:
        print("  Note: cost comparison above is for the volume you specified, but your")
        print("  current PTU count would need to grow to actually handle it without overflow.")
    print()
    print("  Reminder: PTU $/hour, TPM/PTU, and the output-token weight are all")
    print("  configurable inputs here, not fixed facts — Microsoft does not publish a")
    print("  universal number for any of them. Re-run with your own figures once you")
    print("  have a quote and have validated capacity with Azure's own calculator.")
    print()


def estimate_volume_from_docs(
    submissions_per_year: float,
    forms_per_submission: float,
    pages_per_form: float,
    tokens_per_page: float,
    io_ratio: float = 1.0,
) -> dict:
    """Derive monthly input+output token volumes from document/form counts.

    Args:
        submissions_per_year: Number of submissions, claims, or cases per year.
        forms_per_submission: Average number of documents or forms per submission.
        pages_per_form:       Average pages per form/document.
        tokens_per_page:      Estimated tokens per page (800 is a common starting point
                               for dense insurance forms; 500-600 for lightly filled forms).
        io_ratio:             Output tokens as a fraction of input tokens (default 1.0 = 1:1).
                               For extraction-only tasks, 0.1-0.3 is more realistic.
                               For long-form drafting, 0.5-1.5.

    Returns a dict with all intermediate counts and monthly figures.
    """
    total_pages_per_year = submissions_per_year * forms_per_submission * pages_per_form
    total_input_tokens_year = total_pages_per_year * tokens_per_page
    total_output_tokens_year = total_input_tokens_year * io_ratio

    monthly_input = total_input_tokens_year / 12
    monthly_output = total_output_tokens_year / 12

    return {
        "submissions_per_year": submissions_per_year,
        "forms_per_submission": forms_per_submission,
        "pages_per_form": pages_per_form,
        "tokens_per_page": tokens_per_page,
        "io_ratio": io_ratio,
        "total_pages_year": total_pages_per_year,
        "total_input_tokens_year": total_input_tokens_year,
        "total_output_tokens_year": total_output_tokens_year,
        "monthly_input_tokens": monthly_input,
        "monthly_output_tokens": monthly_output,
    }

    print(f"  = {v['total_input_tokens_year']:,.0f} input tokens/year  "
          f"({v['total_input_tokens_year'] / 1e6:.1f}M)")
    print(f"  ÷ 12 months  = {v['monthly_input_tokens']:,.0f} input tokens/month  "
          f"({v['monthly_input_tokens'] / 1e6:.1f}M/month)")
    print()
    print(f"  Output: {v['io_ratio']:g}× input ratio "
          f"→ {v['monthly_output_tokens']:,.0f} output tokens/month  "
          f"({v['monthly_output_tokens'] / 1e6:.1f}M/month)")
    print()


def print_ptu_multi_model_comparison(
    models: list,
    cfg_template: "PTUConfig",
    monthly_input_tokens: float,
    monthly_output_tokens: float,
    ptu_cost: float,
) -> None:
    """Print a side-by-side PTU vs PAYG comparison table for multiple models.

    cfg_template provides ptus_available and hours_per_month; per-model tpm_per_ptu
    and output_weight are looked up from defaults unless set on the template.
    """
    print()
    print("-" * 72)
    print(f"MULTI-MODEL PTU vs. PAYG COMPARISON  ({cfg_template.ptus_available:g} PTUs @ "
          f"{fmt_money(ptu_cost)}/hr)")
    print(f"Volume: {monthly_input_tokens:,.0f} input + {monthly_output_tokens:,.0f} output tokens/month")
    print("-" * 72)
    header = f"{'Model':<22}  {'Req TPM':>9}  {'Util%':>6}  {'PTU $/mo':>12}  {'PAYG $/mo':>12}  Cheaper"
    print(header)
    print("-" * 72)
    for m in models:
        cfg = PTUConfig(
            ptus_available=cfg_template.ptus_available,
            ptu_hourly_cost=ptu_cost,
            tpm_per_ptu=cfg_template.tpm_per_ptu,
            hours_per_month=cfg_template.hours_per_month,
            output_weight=cfg_template.output_weight,
        )
        r = compare_ptu_vs_payg(m, cfg, monthly_input_tokens, monthly_output_tokens)
        util_str = f"{r['utilization_pct']:.1f}%" if r["utilization_pct"] is not None else "n/a"
        cheaper = r["cheaper_option"]
        saving = fmt_money(r["monthly_savings"])
        print(f"  {m.name:<20}  {r['required_tpm']:>9,.0f}  {util_str:>6}  "
              f"{fmt_money(r['ptu_monthly_cost']):>12}  {fmt_money(r['payg_monthly_cost']):>12}  "
              f"{cheaper} (save {saving}/mo)")
    print("-" * 72)
    print("  PTU $/PTU/hr and TPM/PTU defaults are planning estimates — verify before committing.")
    print()

def fmt_money(x: float, decimals: int = 2) -> str:
    return f"${x:,.{decimals}f}"


def explain_runner_up(top: Model, runner_up: Model) -> str:
    if (runner_up.intelligence or 0) > (top.intelligence or 0):
        return "higher accuracy"
    if runner_up.input_price < top.input_price:
        return "lower cost at scale"
    if runner_up.context > top.context:
        return "a larger context window"
    return "a different latency profile"


def print_recommendation(uc: UseCase, top_n: int = 5, explain: bool = False) -> None:
    ranked = rank_models(uc)
    top_model, top_score = ranked[0]
    runner_up, _ = ranked[1]

    print()
    print("=" * 72)
    print(f"USE CASE: {uc.name}")
    if uc.category != "Ad hoc":
        print(f"Category: {uc.category}")
    if uc.note:
        print(f"Note: {uc.note}")
    print("=" * 72)

    flags = []
    if uc.regulated:
        flags.append("regulated/auditable")
    if uc.vision:
        flags.append("vision/document input")
    if uc.audio:
        flags.append("native audio input")
    if uc.video:
        flags.append("native video input")
    if uc.image_gen:
        flags.append("image generation")
    if uc.open_weight:
        flags.append("needs open weights / self-hosting")
    flag_str = ", ".join(flags) if flags else "none"
    print(f"Profile: complexity={uc.complexity}/5  latency={uc.latency}/5  "
          f"volume={uc.volume}/5  context={uc.context}/5  flags=[{flag_str}]")
    print()

    print(f"RECOMMENDED: {top_model.name}  ({top_model.vendor})")
    intel_str = str(top_model.intelligence) if top_model.intelligence is not None else "n/a"
    print(f"  Intelligence Index: {intel_str}   "
          f"Context: {top_model.context:,} tokens   "
          f"Open weights: {'yes' if top_model.open_weight else 'no'}")
    print(f"  Price: {fmt_money(top_model.input_price)}/M in, "
          f"{fmt_money(top_model.output_price)}/M out, "
          f"blended {fmt_money(top_model.blended_price())}/M (7:2:1 cache:in:out)")
    print(f"  Speed: {top_model.speed} tok/s   TTFT: {top_model.ttft}s   "
          f"Cutoff: {top_model.knowledge_cutoff}   Released: {top_model.release_date}")
    print()
    print(f"  Runner-up: {runner_up.name} ({runner_up.vendor}) — consider if you need "
          f"{explain_runner_up(top_model, runner_up)}.")

    if uc.regulated:
        print("  Regulated workflow: log prompts/model version/output, keep a human")
        print("  in the loop on edge cases, and document a model card for audit.")
    if uc.complexity == 5 and uc.regulated:
        print("  High complexity + regulated: treat any model's output as analysis")
        print("  feeding an actuary or compliance sign-off — never the decision itself.")

    print()
    print(f"Top {top_n} models for this use case:")
    print(f"  {'Model':<22}{'Vendor':<11}{'In $/M':>8}{'Out $/M':>9}{'Blend':>8}{'Cache':>8}"
          f"{'Context':>11}{'Intel.':>7}{'Speed':>7}{'TTFT':>7}")
    for m, sc in ranked[:top_n]:
        intel_disp = str(m.intelligence) if m.intelligence is not None else "n/a"
        cache_disp = f"{m.cache_hit_price:.3f}" if m.cache_hit_price is not None else "n/a"
        print(f"  {m.name:<22}{m.vendor:<11}{m.input_price:>8.2f}{m.output_price:>9.2f}"
              f"{m.blended_price():>8.3f}{cache_disp:>8}{m.context:>11,}{intel_disp:>7}{m.speed:>7}{m.ttft:>7.1f}")
    print("  (Knowledge cutoff, release date, and eval cost: run --explain for full detail on any model above.)")
    print()

    if explain:
        print_model_justification(top_model)

    ("ttft", "Time to first token"),

    ("output_price", "Output price"),

    ("context", "Context window"),
    ("knowledge_cutoff", "Knowledge cutoff"),
    ("release_date", "Release date"),
    ("eval_cost", "Eval cost"),
    ("open_weight", "Open weights"),
    ("vision", "Vision / document input"),
    ("audio_input", "Native audio input"),
    ("video_input", "Native video input"),
    ("image_gen", "Image generation"),
]


def format_param_value(m: Model, key: str) -> str:
    if key == "intelligence":
        return str(m.intelligence) if m.intelligence is not None else "n/a"
    if key == "speed":
        return f"{m.speed} tok/s"
    if key == "ttft":
        return f"{m.ttft}s"
    if key == "input_price":
        return f"{fmt_money(m.input_price)}/M"
    if key == "output_price":
        return f"{fmt_money(m.output_price)}/M"
    if key == "blended_price":
        return f"{fmt_money(m.blended_price(), 3)}/M"
    if key == "cache_hit_price":
        return f"{fmt_money(m.cache_hit_price, 3)}/M" if m.cache_hit_price is not None else "not published"
    if key == "context":
        return f"{m.context:,} tokens"
    if key == "knowledge_cutoff":
        return m.knowledge_cutoff
    if key == "release_date":
        return m.release_date
    if key == "eval_cost":
        return fmt_money(m.eval_cost) if m.eval_cost is not None else "not published"
    if key == "open_weight":
        return "yes" if m.open_weight else "no"
    if key == "audio_input":
        return "yes" if m.audio_input else "no"
    if key == "video_input":
        return "yes" if m.video_input else "no"
    if key == "image_gen":
        return "yes" if m.image_gen else "no"
    if key == "vision":
        return "yes" if m.vision else "no"
    return "?"


def print_model_justification(m: Model) -> None:
    """Print a full per-parameter justification block for one model."""
    print("-" * 72)
    print(f"WHY THESE NUMBERS — {m.name} ({m.vendor})")
    print("-" * 72)
    if m.notes:
        print(f"  {m.notes}\n")
    for key, label in PARAM_ORDER:
        value = format_param_value(m, key)
        print(f"  {label}: {value}")
        reason = m.justify(key)
        for line in textwrap.wrap(reason, width=86, initial_indent="    ", subsequent_indent="    "):
            print(line)
        print()

def print_explain_model(name: str) -> bool:
    """Print justification for a model looked up by (partial, case-insensitive) name."""
    q = name.lower().strip()
    matches = [m for m in MODELS if q in m.name.lower()]
    if not matches:
        print(f'\nNo model matching "{name}" found. Run --models to see all model names.\n')
        return False
    for m in matches:
        print_model_justification(m)
    return True

BANNER = """
Model Advisor — which LLM should you use for this job?

Describe a use case in plain English (e.g. "FNOL triage" or
"summarize a 200-page commercial submission") and I'll recommend a model,
with reasoning, across Claude, GPT, Gemini, Llama, DeepSeek, and Mistral.

Commands:
  list                 show all known use cases
  models                show the full model table
  ptu                    walk through a PTU vs. pay-as-you-go cost comparison
  explain <model name>   full per-parameter justification for one model
  <use case> +explain    recommendation + justification for the top model
  help                   show this message again
  quit                   exit
"""


def _prompt_float(label: str, default: Optional[float] = None) -> Optional[float]:
    suffix = f" [{default}]" if default is not None else ""
    raw = input(f"{label}{suffix}: ").strip()
    if not raw:
        return default
    try:
        return float(raw.replace(",", ""))
    except ValueError:
        print("  Not a number — using default." if default is not None else "  Not a number — skipping.")
        return default


def run_ptu_wizard() -> None:
    """Interactive walkthrough for the PTU vs. PAYG comparison in chat mode."""
    print("\nPTU vs. pay-as-you-go calculator")
    print("Every PTU-specific number below is yours to set — Azure doesn't publish")
    print("one universal figure for any of them. Press Enter to accept a labeled default.\n")

    # --- Volume source ---
    print("How do you want to specify your volume?")
    print("  1) Monthly token counts directly  (e.g. 80M input : 80M output)")
    print("  2) From document/form stats        (submissions × forms × pages × tokens/page)")
    vol_choice = input("Choose 1-2 [1]: ").strip() or "1"

    if vol_choice == "2":
        print("\nDocument volume inputs:")
        subs = _prompt_float("Submissions (claims/cases/files) per year")
        forms = _prompt_float("Forms (documents) per submission", 1.0)
        pages = _prompt_float("Pages per form", 5.0)
        tpp = _prompt_float("Tokens per page (800 = dense insurance form)", 800.0)
        io_ratio = _prompt_float("Output:input ratio (1.0 = equal in/out; 0.2 = extraction only)", 1.0)
        if subs is None:
            print("Submissions/year is required.\n")
            return
        v = estimate_volume_from_docs(subs, forms or 1.0, pages or 5.0, tpp or 800.0, io_ratio or 1.0)
        print_doc_volume_derivation(v)
        monthly_in = v["monthly_input_tokens"]
        monthly_out = v["monthly_output_tokens"]
    else:
        monthly_in = _prompt_float("Expected monthly INPUT tokens")
        monthly_out = _prompt_float("Expected monthly OUTPUT tokens")
        if monthly_in is None or monthly_out is None:
            print("Monthly token volume is required.\n")
            return

    # --- Multi-model or single ---
    print("\nCompare against:")
    print("  1) A single model")
    print("  2) Multiple models side-by-side")
    model_choice = input("Choose 1-2 [1]: ").strip() or "1"

    if model_choice == "2":
        raw = input("Model names (comma-separated, partial matches OK): ").strip()
        model_list = []
        for name in raw.split(","):
            m = find_model_by_name(name.strip())
            if m:
                model_list.append(m)
            else:
                print(f'  (skipping — no model matching "{name.strip()}" found)')
        if not model_list:
            print("No valid models found.\n")
            return
    else:
        model_query = input("Model (name or partial match): ").strip()
        model = find_model_by_name(model_query)
        if model is None:
            print(f'No model matching "{model_query}" found. Run "models" to see all names.\n')
            return
        model_list = [model]

    ptus = _prompt_float("Number of PTUs you have or are pricing out")
    if ptus is None:
        print("PTU count is required.\n")
        return

    print("\nBilling commitment affects the default $/PTU/hour:")
    print("  1) hourly                ~$3.35/PTU/hr")
    print("  2) monthly reservation   ~$2.02/PTU/hr")
    print("  3) annual reservation    ~$1.31/PTU/hr")
    billing_choice = input("Choose 1-3 [1]: ").strip() or "1"
    billing_map = {"1": "hourly", "2": "monthly_reservation", "3": "annual_reservation"}
    billing = billing_map.get(billing_choice, "hourly")
    default_cost = PTU_COST_DEFAULTS[billing]
    ptu_cost = _prompt_float("$ per PTU per hour", default_cost)

    cfg_template = PTUConfig(ptus_available=ptus, ptu_hourly_cost=ptu_cost or default_cost)

    if len(model_list) == 1:
        model = model_list[0]
        default_tpm = TPM_PER_PTU_DEFAULTS.get(model.name, DEFAULT_TPM_PER_PTU_FALLBACK)
        tpm_per_ptu = _prompt_float(f"Tokens/minute per PTU for {model.name} (planning default shown)", default_tpm)
        default_weight = round(model.output_price / model.input_price, 2) if model.input_price else 4.0
        output_weight = _prompt_float("Output-token capacity weight (derived from price ratio, shown as default)", default_weight)
        cfg = PTUConfig(ptus_available=ptus, ptu_hourly_cost=ptu_cost or default_cost,
                        tpm_per_ptu=tpm_per_ptu, output_weight=output_weight)
        print_ptu_comparison(model, cfg, monthly_in, monthly_out)
    else:
        print_ptu_multi_model_comparison(model_list, cfg_template, monthly_in, monthly_out, ptu_cost or default_cost)


def run_chat() -> None:
    print(textwrap.dedent(BANNER))
    while True:
        try:
            query = input("Use case > ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nGoodbye.")
            break

        if not query:
            continue
        low = query.lower()
        if low in ("quit", "exit", "q"):
            print("Goodbye.")
            break
        if low in ("help", "?"):
            print(textwrap.dedent(BANNER))
            continue
        if low in ("list", "use cases", "usecases"):
            print_use_case_list()
            continue
        if low in ("models", "model table"):
            print_model_table()
            continue
        if low in ("ptu", "ptus", "ptu calculator"):
            run_ptu_wizard()
            continue
        if low.startswith("explain "):
            print_explain_model(query[8:].strip())
            continue

        explain = False
        if low.endswith("+explain"):
            explain = True
            query = query[: -len("+explain")].strip()

        uc = find_use_case(query)
        if uc is None:
            uc = guess_use_case_from_description(query)
        print_recommendation(uc, explain=explain)

def main() -> None:
    parser = argparse.ArgumentParser(
        description="Recommend an LLM for a given (insurance) use case, with full "
                     "per-parameter justification available via --explain.",
    )
    parser.add_argument("query", nargs="*", help="Use case description (omit for chat mode)")
    parser.add_argument("--list", action="store_true", help="List all known use cases and exit")
    parser.add_argument("--models", action="store_true", help="Show the full model table and exit")
    parser.add_argument("--top", type=int, default=5, help="How many ranked models to show (default 5)")
    parser.add_argument("--explain", nargs="?", const="__TOP__", default=None, metavar="MODEL",
                         help="Show full per-parameter justification. With a use case query, "
                              "explains the top-recommended model. Used alone (or with --models), "
                              "explains a specific model by name, e.g. --explain 'Claude Sonnet 4.6'.")
    ptu_group = parser.add_argument_group(
        "PTU vs. pay-as-you-go (Azure Provisioned Throughput)",
        "Compare reserved PTU capacity against per-token pricing for a model. "
        "Requires --ptu-model, --ptus, and --monthly-tokens at minimum.")
    ptu_group.add_argument("--ptu-model", metavar="MODEL", help="Model to run the PTU comparison for")
    ptu_group.add_argument("--ptus", type=float, help="Number of PTUs you have or are pricing out")
    ptu_group.add_argument("--ptu-cost", type=float, default=None,
                            help="$ per PTU per hour (default: industry-reported hourly rate, ~$3.35)")
    ptu_group.add_argument("--ptu-billing", choices=["hourly", "monthly_reservation", "annual_reservation"],
                            default="hourly", help="Billing commitment, used only if --ptu-cost is not set")
    ptu_group.add_argument("--tpm-per-ptu", type=float, default=None,
                            help="Tokens/minute one PTU delivers for this model (default: a rough "
                                 "per-model planning estimate — verify with Azure's own calculator)")
    ptu_group.add_argument("--output-weight", type=float, default=None,
                            help="How many input-token-equivalents one output token consumes against "
                                 "PTU capacity (default: derived from the model's output:input price ratio)")
    ptu_group.add_argument("--monthly-tokens", type=str, metavar="INPUT:OUTPUT",
                            help="Expected monthly volume as 'input:output', e.g. '50000000:10000000'")
    ptu_group.add_argument("--submissions", type=float, default=None, metavar="N",
                            help="Submissions (claims/cases/files) per year — alternative to --monthly-tokens")
    ptu_group.add_argument("--forms-per-submission", type=float, default=1.0, metavar="N",
                            help="Average forms/documents per submission (default: 1)")
    ptu_group.add_argument("--pages-per-form", type=float, default=5.0, metavar="N",
                            help="Average pages per form (default: 5)")
    ptu_group.add_argument("--tokens-per-page", type=float, default=800.0, metavar="N",
                            help="Estimated tokens per page (default: 800 — dense insurance form)")
    ptu_group.add_argument("--io-ratio", type=float, default=1.0, metavar="RATIO",
                            help="Output:input token ratio (default: 1.0; use 0.2 for extraction-only)")
    ptu_group.add_argument("--compare-models", type=str, metavar="MODEL1,MODEL2,...",
                            help="Comma-separated model names for side-by-side multi-model PTU comparison")
    args = parser.parse_args()

    if args.ptu_model or args.compare_models:
        # ---- Resolve volume (tokens directly or from doc stats) ----
        if args.submissions is not None:
            v = estimate_volume_from_docs(
                args.submissions,
                args.forms_per_submission,
                args.pages_per_form,
                args.tokens_per_page,
                args.io_ratio,
            )
            print_doc_volume_derivation(v)
            in_tok = v["monthly_input_tokens"]
            out_tok = v["monthly_output_tokens"]
        elif args.monthly_tokens:
            try:
                in_tok, out_tok = (float(x) for x in args.monthly_tokens.split(":"))
            except ValueError:
                print('\n--monthly-tokens must be in the form INPUT:OUTPUT, e.g. "50000000:10000000"\n')
                return
        else:
            print("\nProvide volume via --monthly-tokens INPUT:OUTPUT or via "
                  "--submissions / --forms-per-submission / --pages-per-form / --tokens-per-page\n")
            return

        ptu_cost = args.ptu_cost if args.ptu_cost is not None else PTU_COST_DEFAULTS[args.ptu_billing]

        # ---- Multi-model comparison ----
        if args.compare_models:
            model_names = [n.strip() for n in args.compare_models.split(",")]
            model_list = []
            for name in model_names:
                m = find_model_by_name(name)
                if m:
                    model_list.append(m)
                else:
                    print(f'  (skipping — no model matching "{name}" found)')
            if args.ptu_model:
                m = find_model_by_name(args.ptu_model)
                if m and m not in model_list:
                    model_list.insert(0, m)
            if not model_list:
                print('\nNo valid models found. Run --models to see all model names.\n')
                return
            if args.ptus is None:
                print("\n--ptus is required for multi-model PTU comparison.\n")
                return
            cfg_template = PTUConfig(ptus_available=args.ptus, ptu_hourly_cost=ptu_cost,
                                     tpm_per_ptu=args.tpm_per_ptu, output_weight=args.output_weight)
            print_ptu_multi_model_comparison(model_list, cfg_template, in_tok, out_tok, ptu_cost)
            return

        # ---- Single-model comparison ----
        model = find_model_by_name(args.ptu_model)
        if model is None:
            print(f'\nNo model matching "{args.ptu_model}" found. Run --models to see all model names.\n')
            return
        if args.ptus is None:
            print("\n--ptu-model requires --ptus. See --help for all PTU options.\n")
            return
        cfg = PTUConfig(ptus_available=args.ptus, ptu_hourly_cost=ptu_cost,
                        tpm_per_ptu=args.tpm_per_ptu, output_weight=args.output_weight)
        print_ptu_comparison(model, cfg, in_tok, out_tok)
        return

    if args.list:
        print_use_case_list()
        return

    if args.explain is not None and not args.query:
        if args.explain == "__TOP__":
            print_model_table()
        else:
            print_explain_model(args.explain)
        return

    if args.models:
        print_model_table()
        return

    if args.query:
        query = " ".join(args.query)
        uc = find_use_case(query) or guess_use_case_from_description(query)
        explain = args.explain is not None
        print_recommendation(uc, top_n=args.top, explain=explain)
        return

    run_chat()


if __name__ == "__main__":
    main()
