function scoreModel(m, uc) {
  let s = 0;
  const intel = m.intelligence !== null ? m.intelligence : INTEL_TARGET[3];
  const target = INTEL_TARGET[uc.complexity];
  s -= Math.abs(intel - target) * 1.3;
  if (intel < target - 10) s -= 25;

  s -= m.ttft * 0.8;
  if (uc.latency >= 4) s += Math.min(m.speed / 4, 40) - m.ttft * 3;

  if (uc.volume >= 4) s -= (m.output_price / 2 + m.input_price) * 2.5;

  const ctxNeed = CONTEXT_FLOOR[uc.context];
  if (m.context < ctxNeed) s -= 45;
  else s += Math.min((m.context - ctxNeed) / 500000, 6);

  if (uc.regulated && !m.reasoning && uc.complexity >= 3) s -= 10;
  if (uc.regulated && m.vendor === "Anthropic") s += 6;

  if (uc.vision) {
    if (!m.vision) s -= 100;
    else if (m.ocr_accuracy) s += (m.ocr_accuracy - 85) * 2.2;
    else s -= 6;
  }

  if (uc.open_weight && !m.open_weight) s -= 60;
  if (uc.audio && !m.audio_input) s -= 120;
  if (uc.video && !m.video_input) s -= 120;
  if (uc.image_gen && !m.image_gen) s -= 80;

  return s;
}

function scoreBreakdown(m, uc) {
  const lines = [];
  const intel = m.intelligence !== null ? m.intelligence : INTEL_TARGET[3];
  const target = INTEL_TARGET[uc.complexity];
  const intelPenalty = -Math.abs(intel - target) * 1.3;
  lines.push({ why: `Intelligence ${m.intelligence ?? "n/a"} vs. target ${target} for complexity ${uc.complexity}/5`, pts: intelPenalty });
  if (intel < target - 10) lines.push({ why: `Significantly under-provisioned for this complexity level`, pts: -25 });

  lines.push({ why: `Baseline time-to-first-token cost (${m.ttft}s)`, pts: -m.ttft * 0.8 });
  if (uc.latency >= 4) {
    const speedBonus = Math.min(m.speed / 4, 40);
    const ttftPenalty = -m.ttft * 3;
    lines.push({ why: `Real-time latency need: speed bonus (${m.speed} tok/s)`, pts: speedBonus });
    lines.push({ why: `Real-time latency need: extra TTFT penalty`, pts: ttftPenalty });
  }

  if (uc.volume >= 4) {
    const volPenalty = -(m.output_price / 2 + m.input_price) * 2.5;
    lines.push({ why: `High-volume cost weighting on price`, pts: volPenalty });
  }

  const ctxNeed = CONTEXT_FLOOR[uc.context];
  if (m.context < ctxNeed) {
    lines.push({ why: `Context window (${(m.context/1000).toLocaleString()}K) below the ${(ctxNeed/1000).toLocaleString()}K floor needed`, pts: -45 });
  } else {
    const bonus = Math.min((m.context - ctxNeed) / 500000, 6);
    lines.push({ why: `Context headroom above the ${(ctxNeed/1000).toLocaleString()}K floor needed`, pts: bonus });
  }

  if (uc.regulated && !m.reasoning && uc.complexity >= 3) {
    lines.push({ why: `Regulated workflow + non-reasoning model at complexity 3+`, pts: -10 });
  }
  if (uc.regulated && m.vendor === "Anthropic") {
    lines.push({ why: `Regulated workflow: Anthropic benchmarked-low-hallucination bonus`, pts: 6 });
  }

  if (uc.vision) {
    if (!m.vision) {
      lines.push({ why: `Document/image input required but not supported`, pts: -100 });
    } else if (m.ocr_accuracy) {
      lines.push({ why: `Measured OCR accuracy bonus (${m.ocr_accuracy}%)`, pts: (m.ocr_accuracy - 85) * 2.2 });
    } else {
      lines.push({ why: `Vision-capable but no published OCR accuracy figure`, pts: -6 });
    }
  }

  if (uc.open_weight && !m.open_weight) lines.push({ why: `Open weights / self-hosting required but unavailable`, pts: -60 });
  if (uc.audio && !m.audio_input) lines.push({ why: `Native audio input required but not supported`, pts: -120 });
  if (uc.video && !m.video_input) lines.push({ why: `Native video input required but not supported`, pts: -120 });
  if (uc.image_gen && !m.image_gen) lines.push({ why: `Image generation required but not supported`, pts: -80 });

  return lines;
}

function rankModels(uc) {
  return MODELS.map(m => ({ model: m, score: scoreModel(m, uc) }))
    .sort((a, b) => b.score - a.score);
}

function findUseCase(query) {
  const q = query.toLowerCase().trim();
  if (!q) return null;
  for (const uc of USE_CASES) {
    if (uc.name.toLowerCase() === q) return uc;
  }
  let best = null, bestScore = 0;
  for (const uc of USE_CASES) {
    const haystack = [uc.name.toLowerCase(), ...uc.keywords];
    let score = 0;
    for (const term of haystack) {
      if (q.includes(term) || term.includes(q)) score += term.length;
    }
    if (score > bestScore) { best = uc; bestScore = score; }
  }
  return best;
}

  if (uc.vision) flags.push("vision/document input");
  if (uc.audio) flags.push("native audio input");
  if (uc.video) flags.push("native video input");
  if (uc.image_gen) flags.push("image generation");
  if (uc.open_weight) flags.push("needs open weights / self-hosting");

  let warningHtml = "";
  const allExcluded = ranked.every(r => r.score < -90);
  if (uc.audio && !top.audio_input) {
    warningHtml = `<div class="verdict-warning">⚠ No model in this table fully satisfies the native-audio requirement at a high score — check the ledger below before committing.</div>`;
  }

  const html = `
    <div class="verdict">
      <div class="verdict-label">${uc.category !== "Ad hoc" ? uc.category : "Custom profile"} · Recommended</div>
      <div class="verdict-name">${top.name}</div>
      <div class="verdict-vendor">${top.vendor}</div>
      <div class="verdict-meta">
        <div class="m"><span class="n">${top.intelligence ?? "n/a"}</span><span class="l">Intelligence Index</span></div>
        <div class="m"><span class="n">${(top.context/1000).toLocaleString()}K</span><span class="l">Context</span></div>
        <div class="m"><span class="n">${fmtMoney(top.blended_price,3)}/M</span><span class="l">Blended price</span></div>

        <div class="m"><span class="n">${top.ttft}s</span><span class="l">TTFT</span></div>
      </div>
      <div class="verdict-note">
        <b>Profile:</b> complexity ${uc.complexity}/5 · latency ${uc.latency}/5 · volume ${uc.volume}/5 · context ${uc.context}/5
        ${flags.length ? " · " + flags.join(", ") : ""}
        <br><br>
        <b>Runner-up:</b> ${runnerUp.name} (${runnerUp.vendor}) — consider if you need ${runnerReason}.
        ${uc.regulated ? "<br><br>Regulated workflow: log prompts/model version/output, keep a human in the loop on edge cases, document a model card for audit." : ""}
        ${uc.complexity === 5 && uc.regulated ? "<br><br>High complexity + regulated: treat any model's output as analysis feeding a sign-off — never the decision itself." : ""}
      </div>
      ${warningHtml}
      <div class="ledger-toggle" id="ledgerToggle"><span class="arrow">▸</span> Show full scoring ledger (all 20 models)</div>
    </div>
    <div class="ledger" id="ledgerBody"></div>
  `;
  els.verdictArea.innerHTML = html;

  const ledgerToggle = document.getElementById("ledgerToggle");
  const ledgerBody = document.getElementById("ledgerBody");
  ledgerToggle.onclick = () => {
    ledgerToggle.classList.toggle("open");
    ledgerBody.classList.toggle("open");
    if (ledgerBody.classList.contains("open") && !ledgerBody.dataset.built) {
      buildLedger(ledgerBody, ranked, uc);
      ledgerBody.dataset.built = "1";
    }
  };
}

function buildLedger(container, ranked, uc) {
  container.innerHTML = "";
  ranked.forEach((entry, i) => {
    const m = entry.model;
    const row = document.createElement("div");
    row.className = "ledger-row";
    const head = document.createElement("div");
    head.className = "ledger-head";
    head.innerHTML = `
      <div><span class="ledger-rank">#${i+1}</span><span class="ledger-name">${m.name}</span><span class="ledger-vendor">${m.vendor}</span></div>
      <div class="ledger-score ${i===0?'top':''}">${entry.score.toFixed(1)} pts</div>
    `;
    const detail = document.createElement("div");
    detail.className = "ledger-detail";
    const lines = scoreBreakdown(m, uc);
    detail.innerHTML = lines.map(l => `
      <div class="ledger-line">
        <span class="why">${l.why}</span>
        <span class="pts ${l.pts >= 0 ? 'pos' : 'neg'}">${l.pts >= 0 ? '+' : ''}${l.pts.toFixed(1)}</span>
      </div>
    `).join("");
    head.onclick = () => detail.classList.toggle("open");
    row.appendChild(head);
    row.appendChild(detail);
    container.appendChild(row);
  });
}

// ============================================================
// PTU vs. pay-as-you-go (Azure Provisioned Throughput) calculator
// Ported 1:1 from model_advisor.py's compare_ptu_vs_payg(). PTU $/hour,
// TPM/PTU, and the output-token capacity weight are all user-configurable —
// Azure does not publish one universal figure for any of them.
// ============================================================

const PTU_COST_DEFAULTS = { hourly: 3.35, monthly_reservation: 2.02, annual_reservation: 1.31 };
const TPM_PER_PTU_DEFAULTS = {
  "Claude Haiku 4.5": 2500, "Claude Sonnet 4.6": 2000, "Claude Opus 4.7": 1200,
  "GPT-5.4 mini": 3000, "GPT-5.4": 2500, "GPT-5.5": 2200,
  "Gemini 3.1 Flash-Lite": 3500, "Gemini 3.1 Pro": 2200,
  "Llama 4 Scout": 2800, "Llama 4 Maverick": 2400,
  "DeepSeek V3.2": 2600, "DeepSeek R1": 2000,
  "Mistral Large 3": 2300, "Phi-4": 4000,
  "Command A": 2400, "Command A+": 2600,
  "Qwen3.7 Max": 2200, "Grok 4.3": 2000,
  "Nova Pro": 2400, "Nova Micro": 4000,
};
const DEFAULT_TPM_PER_PTU_FALLBACK = 2500;

function findModelByName(query) {
  const q = query.toLowerCase().trim();
  let m = MODELS.find(m => m.name.toLowerCase() === q);
  if (m) return m;
  m = MODELS.find(m => m.name.toLowerCase().startsWith(q));
  if (m) return m;
  return MODELS.find(m => m.name.toLowerCase().includes(q)) || null;
}

function comparePtuVsPayg(model, cfg, monthlyIn, monthlyOut) {
  const tpmPerPtu = cfg.tpmPerPtu ?? (TPM_PER_PTU_DEFAULTS[model.name] ?? DEFAULT_TPM_PER_PTU_FALLBACK);
  const outWeight = cfg.outputWeight ?? (model.input_price ? model.output_price / model.input_price : 4.0);

  const weightedTokens = monthlyIn + monthlyOut * outWeight;
  const minutesPerMonth = cfg.hoursPerMonth * 60;
  const requiredTpm = minutesPerMonth ? weightedTokens / minutesPerMonth : 0;
  const availableTpm = tpmPerPtu * cfg.ptusAvailable;
  const coversVolume = availableTpm >= requiredTpm;
  const utilizationPct = availableTpm ? (requiredTpm / availableTpm) * 100 : null;

  const ptuCost = cfg.ptusAvailable * cfg.ptuHourlyCost * cfg.hoursPerMonth;
  const paygCost = (monthlyIn / 1e6) * model.input_price + (monthlyOut / 1e6) * model.output_price;
  const ptusNeeded = requiredTpm ? requiredTpm / tpmPerPtu : 0;

  let cheaper = ptuCost < paygCost ? "PTU" : "PAYG";
  let savings = Math.abs(ptuCost - paygCost);
  let savingsPct = Math.max(ptuCost, paygCost) ? (savings / Math.max(ptuCost, paygCost)) * 100 : 0;

  if (cfg.ptusAvailable <= 0) {
    cheaper = "PAYG";
    savings = paygCost;
    savingsPct = paygCost > 0 ? 100 : 0;
  }

  return {
    requiredTpm, availableTpm, coversVolume, utilizationPct, ptusNeeded,
    ptuCost, paygCost, cheaper, savings, savingsPct, outWeight, tpmPerPtu,
  };
}

function ptuDefaultsForModel(modelName) {
  const m = findModelByName(modelName);
  if (!m) return;
  ptuEls.tpm.placeholder = String(TPM_PER_PTU_DEFAULTS[m.name] ?? DEFAULT_TPM_PER_PTU_FALLBACK);
  ptuEls.weight.placeholder = (m.input_price ? (m.output_price / m.input_price).toFixed(2) : "4.00");
}

function fmtMoneyFull(x, decimals = 2) { return "$" + Number(x).toLocaleString(undefined, { minimumFractionDigits: decimals, maximumFractionDigits: decimals }); }

function updatePtu() {
  const m = findModelByName(ptuEls.model.value);
  if (!m) { ptuEls.result.innerHTML = `<p class="empty-state">Pick a model to compare.</p>`; return; }

  const ptusAvailable = parseFloat(ptuEls.ptus.value) || 0;
  const billing = ptuEls.billing.value;
  const ptuHourlyCost = ptuEls.cost.value !== "" ? parseFloat(ptuEls.cost.value) : PTU_COST_DEFAULTS[billing];
  const tpmPerPtu = ptuEls.tpm.value !== "" ? parseFloat(ptuEls.tpm.value) : null;
  const outputWeight = ptuEls.weight.value !== "" ? parseFloat(ptuEls.weight.value) : null;
  const monthlyIn = parseFloat(ptuEls.monthlyIn.value) || 0;
  const monthlyOut = parseFloat(ptuEls.monthlyOut.value) || 0;

  const cfg = { ptusAvailable, ptuHourlyCost, tpmPerPtu, outputWeight, hoursPerMonth: 730 };
  const r = comparePtuVsPayg(m, cfg, monthlyIn, monthlyOut);

  const tpmUsedLabel = tpmPerPtu !== null ? "your value" : "planning default — verify with Azure's capacity calculator";
  const weightUsedLabel = outputWeight !== null ? "your value" : "derived from output:input price ratio — approximation";

  let coverageLine;
  if (ptusAvailable <= 0) {
    coverageLine = `<span class="pts neg">0 PTUs provides no capacity — not a real deployment option, just a reference point.</span>`;
  } else if (r.coversVolume) {
    coverageLine = `Your PTUs <b>cover</b> this volume, at <b>${r.utilizationPct.toFixed(1)}%</b> utilization.`;
  } else {
    const utilStr = r.utilizationPct !== null ? r.utilizationPct.toFixed(1) + "%" : "undefined";
    coverageLine = `Your PTUs <b>do not cover</b> this volume (${utilStr} of capacity needed). You'd need ~<b>${Math.max(r.ptusNeeded,0.1).toFixed(1)} PTUs</b> to fully cover it.`;
  }

  let costLine;
  if (ptusAvailable <= 0) {
    costLine = `With 0 PTUs there's no PTU option — PAYG is the only choice at <b>${fmtMoneyFull(r.paygCost)}/month</b>.`;
  } else {
    costLine = `<b>${r.cheaper}</b> is cheaper by <b>${fmtMoneyFull(r.savings)}/month</b> (${r.savingsPct.toFixed(1)}%) at this volume.`;
  }

  let warnLine = "";
  if (r.utilizationPct !== null && r.utilizationPct < 50 && r.cheaper === "PTU" && ptusAvailable > 0) {
    warnLine = `<div class="verdict-warning">⚠ Utilization is under 50% — PTU often only wins at sustained high utilization. Re-check once you have real traffic data.</div>`;
  }

  ptuEls.result.innerHTML = `
    <div class="verdict">
      <div class="verdict-label">PTU vs. pay-as-you-go · ${m.name}</div>
      <div class="verdict-meta" style="margin-top:14px;">
        <div class="m"><span class="n">${fmtMoneyFull(r.ptuCost)}</span><span class="l">PTU cost / month</span></div>
        <div class="m"><span class="n">${fmtMoneyFull(r.paygCost)}</span><span class="l">PAYG cost / month</span></div>
        <div class="m"><span class="n">${r.requiredTpm.toLocaleString(undefined,{maximumFractionDigits:0})}</span><span class="l">TPM required</span></div>
        <div class="m"><span class="n">${r.availableTpm.toLocaleString(undefined,{maximumFractionDigits:0})}</span><span class="l">TPM available</span></div>
      </div>
      <div class="verdict-note">
        ${coverageLine}<br><br>
        ${costLine}<br><br>
        <span style="color:#8E94A8;">TPM/PTU used: ${r.tpmPerPtu.toLocaleString()} (${tpmUsedLabel}) · Output weight used: ${r.outWeight.toFixed(2)}x (${weightUsedLabel})</span>
      </div>
      ${warnLine}
    </div>
  `;
}
